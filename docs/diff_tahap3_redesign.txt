+def get_dir_fingerprint(root_dir: Path) -> dict:
+    """Menghitung hash sha256 dan ukuran untuk setiap file dalam direktori."""
+    fingerprint = {}
+    if not root_dir.exists():
+        return fingerprint
+    for path in sorted(root_dir.rglob("*")):
+        if path.is_file():
+            rel = str(path.relative_to(root_dir)).replace("\\", "/")
+            h = hashlib.sha256(path.read_bytes()).hexdigest()
+            fingerprint[rel] = (path.stat().st_size, h)
+    return fingerprint
+

+def run_single_benchmark_turn_set3(
+    prompt_item: Dict[str, Any],

+    """Menjalankan 1 prompt dalam ReAct loop multi-turn untuk Set 3."""

+    tool_calls_recorded = []

+
+    def trace_cb(evt: str, msg: str):
